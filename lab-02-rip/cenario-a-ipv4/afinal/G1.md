#### Conexão das interfaces:

> G0/0 -> Trunk para SwitchCentral (inter-VLAN routing)
> G0/1 -> Conecta com o RX com menor X
> G0/2 -> Conecta com o RX com maior X

### Switch Central

```ios
enable
conf t
hostname SwitchCentral
no ip domain-lookup
enable secret class
banner motd # ! Acesso restrito, continue apenas se autorizado ! #
line console 0
 password class
 login
 exit
line vty 0 15
 password class
 login
 exit

vlan 2
 name G2
 exit
vlan 3
 name G3
 exit
vlan 4
 name G4
 exit
vlan 5
 name G5
 exit
vlan 6
 name G6
 exit

! Porta trunk para R1_G1
int g0/1
 switchport mode trunk
 exit

! Portas access para cada grupo
int f0/1
 switchport mode access
 switchport access vlan 2
 exit
int f0/2
 switchport mode access
 switchport access vlan 3
 exit
int f0/3
 switchport mode access
 switchport access vlan 4
 exit
int f0/4
 switchport mode access
 switchport access vlan 5
 exit
int f0/5
 switchport mode access
 switchport access vlan 6
 exit

end
copy running-config startup-config
```

### Router 1

```ios
enable
conf t
hostname RouterR1_G1
no ip domain-lookup
enable secret class
banner motd # ! Acesso restrito, continue apenas se autorizado ! #
line console 0
 password class
 login
 exit
line vty 0 4
 password class
 login
 exit

int Loopback0
 description R1 - Rede Interna (Loopback)
 ip address 192.168.13.193 255.255.255.224
 no shut
 exit

! Subinterfaces para inter-VLAN routing
int g0/0
 no ip address
 no shut
 exit

int g0/0.2
 description Inter-VLAN G2
 encapsulation dot1Q 2
 ip address 10.0.0.5 255.255.255.252
 exit

int g0/0.3
 description Inter-VLAN G3
 encapsulation dot1Q 3
 ip address 10.0.0.9 255.255.255.252
 exit

int g0/0.4
 description Inter-VLAN G4
 encapsulation dot1Q 4
 ip address 10.0.0.13 255.255.255.252
 exit

int g0/0.5
 description Inter-VLAN G5
 encapsulation dot1Q 5
 ip address 10.0.0.17 255.255.255.252
 exit

int g0/0.6
 description Inter-VLAN G6
 encapsulation dot1Q 6
 ip address 10.0.0.21 255.255.255.252
 exit

int g0/1
 description R1 - Rede p/ R2
 ip address 192.168.13.225 255.255.255.252
 no shut
 exit

int g0/2
 description R1 - Rede p/ R3
 ip address 192.168.13.229 255.255.255.252
 no shut
 exit

! --- RIPv2 Configuration ---
router rip
version 2
no auto-summary
passive-interface Loopback0
network 192.168.13.192
network 192.168.13.224
network 192.168.13.228
network 10.0.0.0
exit

end
copy running-config startup-config
```
