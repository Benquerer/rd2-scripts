### Router 1

```ios
! --- Basic Configuration ---
enable
conf t
hostname RouterR1_G5
no ip domain-lookup
enable secret class
banner motd # ! Acesso restrito, continue apenas se autorizado ! #
line console 0
 password class
 login
 exit
line vty 0 4
 password class
 login
 exit
int Loopback0
 description R1 - Rede Interna (Loopback)
 ip address 192.168.53.193 255.255.255.224
 no shut
 exit
int g0/0
 description R1 - Ligacao Inter-Grupo p/ G4-R2
 ip address 192.168.11.14 255.255.255.252
 no shut
 exit
int g0/1
 description R1 - Rede p/ R2
 ip address 192.168.53.225 255.255.255.252
 no shut
 exit
int g0/2
 description R1 - Rede p/ R3
 ip address 192.168.53.229 255.255.255.252
 no shut
 exit
router rip
 version 2
 no auto-summary
 passive-interface Loopback0
 network 192.168.53.192
 network 192.168.53.224
 network 192.168.53.228
 network 192.168.11.12
 exit
end
copy running-config startup-config
```

### Router 3

```ios
! --- Basic Configuration ---
enable
conf t
hostname RouterR3_G3
no ip domain-lookup
enable secret class
banner motd # ! Acesso restrito, continue apenas se autorizado ! #

! --- Console Port ---
line console 0
password class
login
exit

! --- Virtual Terminal Lines ---
line vty 0 4
password class
login
exit

! --- Interfaces ---
int g0/0
description R3 - Rede Interna
ip address 192.168.53.129 255.255.255.192
no shut
exit

int g0/1
description R3 - Rede p/ R1
ip address 192.168.53.230 255.255.255.252
no shut
exit

int g0/2
description R3 - Rede p/ R2
ip address 192.168.53.234 255.255.255.252
no shut
exit

! --- RIPv2 Configuration ---
router rip
version 2
no auto-summary
passive-interface GigabitEthernet0/0
network 192.168.53.128
network 192.168.53.228
network 192.168.53.232
exit

! --- Save Configuration ---
end
copy running-config startup-config
```
