## Configurações

### R1

```ios
configure terminal
hostname R1
no ip domain-lookup
enable secret class
banner motd #Acesso autorizado apenas.#
line console 0
 password class
 login
line vty 0 15
 password class
 login
service password-encryption

interface Loopback0
 ip address 1.1.1.1 255.255.255.255

interface f0/0
 ip address 172.100.100.1 255.255.255.0
 no shutdown

interface f0/1
 ip address 1.1.13.1 255.255.255.252
 no shutdown

ip route 1.1.0.0 255.255.0.0 Null0

ip access-list standard FILTER-BGP
 deny 110.110.0.0 0.0.255.255
 permit any

route-map BGP-TO-OSPF permit 10
 match ip address FILTER-BGP

router ospf 1
 router-id 1.1.1.1
 auto-cost reference-bandwidth 1000
 network 1.1.1.1 0.0.0.0 area 0
 network 1.1.13.0 0.0.0.3 area 0
 redistribute bgp 301 subnets route-map BGP-TO-OSPF

route-map SET-LOCAL-PREF permit 10
 set local-preference 200

router bgp 301
 bgp router-id 1.1.1.1
 neighbor 1.1.2.2 remote-as 301
 neighbor 1.1.2.2 update-source Loopback0
 neighbor 1.1.2.2 next-hop-self
 neighbor 172.100.100.100 remote-as 100
 neighbor 172.100.100.100 route-map SET-LOCAL-PREF in
 network 1.1.0.0 mask 255.255.0.0
 maximum-paths 2

end
copy running-config startup-config
```
